Reach the API:
{{- if .Values.gateway.enabled }}
  kubectl get gateway {{ include "agent-kernel.fullname" . }} -n {{ .Release.Namespace }}
{{- else if .Values.serviceLB.enabled }}
  kubectl get service {{ include "agent-kernel.fullname" . }}-io-lb -n {{ .Release.Namespace }}
{{- else }}
  kubectl port-forward -n {{ .Release.Namespace }} service/{{ include "agent-kernel.fullname" . }}-io 8000:80
{{- end }}

Send a chat request:
  curl -s -X POST http://localhost:8000/api/v1/chat \
    -H 'Content-Type: application/json' \
    -d '{"prompt": "Hello", "session_id": "s1", "agent": "<agent>"}'
{{- end }}
